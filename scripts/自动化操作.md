# 开发模式（最常用）
make dev                 # server + web 并行 watch + HMR
make dev-server          # 仅 server
make dev-web             # 仅 web

# 生产模式（后台运行，PID + 日志自动管理）
make start               # build → 后台 node dist/main.js
make stop                # kill 所有 + 端口兜底清理
make restart             # stop + start
make status              # 进程 / 端口 / 数据库 一目了然

# 构建
make build               # shared → server → web
make build-server        # 仅 server
make rebuild             # clean + build

# 数据库
make db-push             # drizzle-kit push 建表
make db-console          # 进 sqlite3 交互终端
make open-db             # 列出全部表 + 行数
make db-reset            # ⚠ 删库（会二次确认）

# 辅助
make clean               # dist / .pids / .logs 全清
make typecheck           # 前后端 tsc --noEmit
make log-tail            # 实时追 server.log
make seed-all            # 调各模块 POST /seed 造演示数据
make init                # install + build + db-push 一键开坑
